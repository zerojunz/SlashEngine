# Exports games: presets, texture forms, icons, building with export/CMakeLists.txt, signing, and the
# export.run and texture.forms commands; builds and checks gameplay code, and searches the
# engine's API. Used by slash-cli and slash-mcp; games never link it.
add_library(slash_exporter
    src/android.cpp
    src/api_search.cpp
    src/android_library.cpp
    src/commands.cpp
    src/cross.cpp
    src/export.cpp
    src/forms.cpp
    src/gameplay.cpp
    src/icons.cpp
    src/ios.cpp
    src/maven.cpp
    src/presets.cpp
    src/sdk.cpp
    src/steps.cpp
    src/toolchains.cpp
    src/xml.cpp)
add_library(slash::exporter ALIAS slash_exporter)

target_include_directories(slash_exporter PUBLIC include)
target_link_libraries(slash_exporter
    PUBLIC slash::command slash::core slash::render slash::runtime
    PRIVATE slash::ai slash::animation slash::audio slash::canvas slash::library slash::navigation slash::platform slash::scene3d slash::shader_compiler slash::text
            slash::texture_encoder slash::ui slash::world)
# Where exports find the engine sources when SLASH_ENGINE_DIR and an installed copy are absent.
target_compile_definitions(slash_exporter PRIVATE SLASH_ENGINE_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
# Wayland 1.18's wayland-scanner, which Linux exports run through slash-cli (cmake/Dependencies.cmake).
if(TARGET slash_wayland_scanner)
    target_link_libraries(slash_exporter PRIVATE slash_wayland_scanner)
    target_compile_definitions(slash_exporter PRIVATE SLASH_WAYLAND_SCANNER)
endif()
slash_target_warnings(slash_exporter)

# What building a game's own code against this engine build takes, for gameplay.build: the
# compiler, flags, include folders and definitions the engine was built with, preloaded into
# cmake/gameplay/CMakeLists.txt's cache with "cmake -C".
set(gameplay_sdk ${CMAKE_BINARY_DIR}/gameplay-sdk.cmake)
set(sdk_content "# Generated by tools/exporter/CMakeLists.txt for cmake/gameplay/CMakeLists.txt.\n")
foreach(variable IN ITEMS CMAKE_CXX_COMPILER CMAKE_MAKE_PROGRAM CMAKE_BUILD_TYPE CMAKE_CXX_FLAGS
        CMAKE_OSX_ARCHITECTURES CMAKE_OSX_DEPLOYMENT_TARGET CMAKE_OSX_SYSROOT)
    string(APPEND sdk_content
        "set(${variable} [==[${${variable}}]==] CACHE STRING \"\" FORCE)\n")
endforeach()
string(APPEND sdk_content
    "set(SLASH_SDK_INCLUDE_DIRECTORIES [==[$<REMOVE_DUPLICATES:$<TARGET_PROPERTY:slash_runtime,INCLUDE_DIRECTORIES>;$<TARGET_PROPERTY:slash_ai,INCLUDE_DIRECTORIES>>]==] CACHE INTERNAL \"\")\n"
    "set(SLASH_SDK_COMPILE_DEFINITIONS [==[$<REMOVE_DUPLICATES:$<TARGET_PROPERTY:slash_runtime,COMPILE_DEFINITIONS>>]==] CACHE INTERNAL \"\")\n")
file(GENERATE OUTPUT ${gameplay_sdk} CONTENT "${sdk_content}" TARGET slash_runtime)
target_compile_definitions(slash_exporter PRIVATE SLASH_GAMEPLAY_SDK="${gameplay_sdk}"
    SLASH_GAMEPLAY_GENERATOR="${CMAKE_GENERATOR}")
