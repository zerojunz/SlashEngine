// The coin_pickup system: calls the functions as Unreal calls CoinPickup.h's, for each entity with
// a CoinPickup.
void run(World& world, double tick_seconds) {
    entt::registry& ecs = world.ecs();
    for (const slash::world::Event& event : world.events()) {
        // ACoinPickup, on the entities made since the last tick.
        if (event.kind == slash::world::kEntityCreated) {
            const entt::entity entity = world.find(event.id);
            if (CoinPickup* self = ecs.try_get<CoinPickup>(entity); self != nullptr) {
                construct(world, entity, *self);
            }
        }
    }
    ecs.view<CoinPickup>().each([&](entt::entity entity, CoinPickup& self) {
        tick(world, entity, self, tick_seconds);
    });
}
