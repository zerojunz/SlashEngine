// The entities a physics event names: the area or body emitting it, and the one entering
// or leaving it.
std::pair<entt::entity, entt::entity> touching(World& world,
                                               const slash::world::Event& event) {
    const slash::json::Value* other =
        event.data.find(event.kind.starts_with("area") ? "area" : "body");
    return {world.find(event.id),
            other != nullptr && other->is_string() ? world.find(other->as_string()) : entt::null};
}

// The frog system: calls the methods as Unity calls Frog.cs's, for each entity with a Frog.
void run(World& world, double tick_seconds) {
    entt::registry& ecs = world.ecs();
    for (const slash::world::Event& event : world.events()) {
        // Start, on the entities made since the last tick.
        if (event.kind == slash::world::kEntityCreated) {
            const entt::entity entity = world.find(event.id);
            if (Frog* self = ecs.try_get<Frog>(entity); self != nullptr) {
                start(world, entity, *self);
            }
        }
        // OnTriggerEnter2D and OnCollisionEnter2D, on both entities, as Unity calls them.
        if (event.kind == "body_entered" || event.kind == "area_entered") {
            const auto [first, second] = touching(world, event);
            const bool trigger =
                event.kind == "area_entered" || ecs.all_of<slash::world::Area2D>(first);
            for (const auto& [entity, other] :
                 {std::pair{first, second}, std::pair{second, first}}) {
                if (Frog* self = ecs.try_get<Frog>(entity); self != nullptr) {
                    if (trigger) {
                        on_trigger_enter2d(world, entity, *self, other);
                    } else {
                        on_collision_enter2d(world, entity, *self, other);
                    }
                }
            }
        }
    }
    ecs.view<Frog>().each([&](entt::entity entity, Frog& self) {
        update(world, entity, self, tick_seconds);
    });
}

    game.add_system("frog", frog::run,
