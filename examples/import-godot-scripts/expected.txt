// The player system: calls the functions as Godot calls player.gd's, for each entity with a Player.
void run(World& world, double tick_seconds) {
    entt::registry& ecs = world.ecs();
    for (const slash::world::Event& event : world.events()) {
        // _ready, on the entities made since the last tick.
        if (event.kind == slash::world::kEntityCreated) {
            const entt::entity entity = world.find(event.id);
            if (Player* self = ecs.try_get<Player>(entity); self != nullptr) {
                ready(world, entity, *self);
            }
        }
        // pressed from coin in godot/main.scene.json, connected to _on_coin_pressed.
        if (event.kind == "pressed" && event.id == "coin") {
            const entt::entity entity = world.find("player");
            if (Player* self = ecs.try_get<Player>(entity); self != nullptr) {
                on_coin_pressed(world, entity, *self, event);
            }
        }
    }
    ecs.view<Player>().each([&](entt::entity entity, Player& self) {
        physics_process(world, entity, self, tick_seconds);
    });
}

} // namespace imported::player

namespace imported {

// Registers player.gd's component and system; register_scripts calls it.
void register_player(slash::world::GameModule& game) {
    game.add_component<player::Player>()
        .set_doc("Walks to its goal and counts the coins it is given.")
        .describe("speed", {.doc = "Pixels a second."});
    game.add_system("player", player::run,
                    {.doc = "Calls the functions of godot/player.gd, ported to C++, as Godot calls "
                            "them."});
}
